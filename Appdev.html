BMI Calculator

React Native · useState · TextInput · Button
▾

Takes weight (kg) and height (cm) from the user, converts height to metres, then computes BMI = weight / height² and displays the result.

import React, { useState } from 'react';
import { View, Text, TextInput, Button, StyleSheet } from 'react-native';

export default function App() {
  const [weight, setWeight] = useState('');
  const [height, setHeight] = useState('');
  const [bmi,    setBmi]    = useState(null);

  const calculateBMI = () => {
    const h = parseFloat(height) / 100; // cm → metres
    const w = parseFloat(weight);
    if (h > 0 && w > 0) setBmi((w / (h * h)).toFixed(2));
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>BMI Calculator</Text>
      <TextInput style={styles.input} placeholder="Weight (kg)"
        keyboardType="numeric" onChangeText={setWeight} />
      <TextInput style={styles.input} placeholder="Height (cm)"
        keyboardType="numeric" onChangeText={setHeight} />
      <Button title="Calculate BMI" onPress={calculateBMI} />
      {bmi && <Text style={styles.result}>Your BMI: {bmi}</Text>}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, padding: 20, justifyContent: 'center' },
  title:     { fontSize: 24, fontWeight: 'bold', marginBottom: 20, textAlign: 'center' },
  input:     { borderWidth: 1, padding: 10, marginBottom: 10, borderRadius: 5 },
  result:    { fontSize: 20, marginTop: 20, textAlign: 'center', color: 'blue' }
});

Expense Manager

React Native · FlatList · Income/Expense tracking
▾

Records income and expense transactions with amounts and categories. Computes running totals for both types using Array.reduce().

import React, { useState } from 'react';
import { View, Text, TextInput, Button, FlatList, StyleSheet } from 'react-native';

export default function App() {
  const [amount,       setAmount]       = useState('');
  const [category,     setCategory]     = useState('');
  const [type,         setType]         = useState('Expense');
  const [transactions, setTransactions] = useState([]);

  const addTransaction = () => {
    setTransactions([...transactions,
      { id: Math.random().toString(), amount: parseFloat(amount), category, type }
    ]);
    setAmount(''); setCategory('');
  };

  const total = (t) => transactions
    .filter(x => x.type === t)
    .reduce((acc, cur) => acc + cur.amount, 0);

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Expense Manager</Text>
      <TextInput style={styles.input} placeholder="Amount"
        keyboardType="numeric" onChangeText={setAmount} value={amount} />
      <TextInput style={styles.input} placeholder="Category"
        onChangeText={setCategory} value={category} />
      <View style={{flexDirection:'row', justifyContent:'space-around', marginBottom:10}}>
        <Button title="Add Income"  onPress={() => { setType('Income');  addTransaction(); }} color="green" />
        <Button title="Add Expense" onPress={() => { setType('Expense'); addTransaction(); }} color="red"   />
      </View>
      <Text>Income: {total('Income')} | Expense: {total('Expense')}</Text>
      <FlatList data={transactions}
        renderItem={({item}) => <Text style={styles.item}>{item.category}: {item.amount} ({item.type})</Text>}
        keyExtractor={item => item.id} />
    </View>
  );
}

Simple Arithmetic Calculator

React Native · + − × ÷ operations
▾

Accepts two numbers and performs addition, subtraction, multiplication, or division based on which button is pressed.

import React, { useState } from 'react';
import { View, Text, TextInput, Button, StyleSheet } from 'react-native';

export default function App() {
  const [num1,   setNum1]   = useState('');
  const [num2,   setNum2]   = useState('');
  const [result, setResult] = useState(null);

  const calculate = (op) => {
    const n1 = parseFloat(num1), n2 = parseFloat(num2);
    const ops = { '+': n1+n2, '-': n1-n2, '*': n1*n2, '/': n1/n2 };
    setResult(ops[op]);
  };

  return (
    <View style={styles.container}>
      <TextInput style={styles.input} placeholder="Number 1"
        keyboardType="numeric" onChangeText={setNum1} />
      <TextInput style={styles.input} placeholder="Number 2"
        keyboardType="numeric" onChangeText={setNum2} />
      <View style={styles.row}>
        {['+','-','*','/'].map(op =>
          <Button key={op} title={op} onPress={() => calculate(op)} />
        )}
      </View>
      {result !== null && <Text style={styles.result}>Result: {result}</Text>}
    </View>
  );
}

Ticket Booking App

React Native · Simulated DB update · Alert
▾

Preloads a list of buses with seat counts. Tapping "Book Now" decrements the seat count (simulating a DB update via state). Shows an alert if the bus is full.

import React, { useState } from 'react';
import { View, Text, Button, FlatList, StyleSheet, Alert } from 'react-native';

export default function App() {
  const [buses, setBuses] = useState([
    { id: '1', route: 'City A to City B', boarding: 'Station 1', seats: 40 },
    { id: '2', route: 'City C to City D', boarding: 'Station 2', seats: 5  },
  ]);

  const bookTicket = (id) =>
    setBuses(buses.map(bus => {
      if (bus.id === id) {
        if (bus.seats > 0) {
          Alert.alert("Success", `Ticket booked for ${bus.route}`);
          return { ...bus, seats: bus.seats - 1 };
        }
        Alert.alert("Sorry", "Bus is full!");
      }
      return bus;
    }));

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Transport Booking</Text>
      <FlatList data={buses}
        renderItem={({item}) => (
          <View style={styles.card}>
            <Text>{item.route} (Boarding: {item.boarding})</Text>
            <Text>Seats Available: {item.seats}</Text>
            <Button title="Book Now" onPress={() => bookTicket(item.id)} />
          </View>
        )}
        keyExtractor={item => item.id} />
    </View>
  );
}

Unit Converter

React Native · km↔mi · kg↔lb
▾

Single input value converted between kilometres/miles and kilograms/pounds depending on which button is tapped.

import React, { useState } from 'react';
import { View, Text, TextInput, Button, StyleSheet } from 'react-native';

export default function App() {
  const [val,    setVal]    = useState('');
  const [result, setResult] = useState('');

  const conversions = {
    kmToMi: n => `${n} km = ${(n * 0.621371).toFixed(2)} miles`,
    miToKm: n => `${n} miles = ${(n * 1.60934).toFixed(2)} km`,
    kgToLb: n => `${n} kg = ${(n * 2.20462).toFixed(2)} lbs`,
    lbToKg: n => `${n} lbs = ${(n * 0.453592).toFixed(2)} kg`,
  };

  const convert = (type) => setResult(conversions[type](parseFloat(val)));

  return (
    <View style={styles.container}>
      <TextInput style={styles.input} placeholder="Enter value"
        keyboardType="numeric" onChangeText={setVal} />
      {['kmToMi','miToKm','kgToLb','lbToKg'].map(t =>
        <Button key={t} title={t} onPress={() => convert(t)} />
      )}
      <Text style={styles.result}>{result}</Text>
    </View>
  );
}

Client-Server Simulation

React Native · setTimeout · ActivityIndicator
▾

Simulates a network request with a 2-second delay using setTimeout. Shows a loading spinner while "waiting" and then renders the JSON response payload.

import React, { useState } from 'react';
import { View, Text, Button, StyleSheet, ActivityIndicator } from 'react-native';

export default function App() {
  const [status,     setStatus]     = useState('Idle');
  const [serverData, setServerData] = useState('');

  const simulateRequest = () => {
    setStatus('Connecting to Server...');
    setServerData('');
    setTimeout(() => {
      setStatus('Response Received');
      setServerData(JSON.stringify({ message: "Hello from Simulated Server!", status: 200 }));
    }, 2000);
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Client-Server Simulation</Text>
      <Text>Status: {status}</Text>
      {status === 'Connecting to Server...' &&
        <ActivityIndicator size="large" color="#0000ff" />}
      <Button title="Send Request to Server" onPress={simulateRequest} />
      {serverData !== '' && (
        <View style={styles.box}>
          <Text>Server Response:</Text>
          <Text style={{color:'green'}}>{serverData}</Text>
        </View>
      )}
    </View>
  );
}

To-Do List Management

React Native · TouchableOpacity · tap-to-delete
▾

Add tasks via text input; tap any task to remove it. Uses TouchableOpacity for the delete interaction.

import React, { useState } from 'react';
import { View, Text, TextInput, Button, FlatList, StyleSheet, TouchableOpacity } from 'react-native';

export default function App() {
  const [task,  setTask]  = useState('');
  const [tasks, setTasks] = useState([]);

  const addTask    = () => {
    if (task.trim()) {
      setTasks([...tasks, { id: Date.now().toString(), text: task }]);
      setTask('');
    }
  };
  const removeTask = (id) => setTasks(tasks.filter(t => t.id !== id));

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Day-to-Day To-Do</Text>
      <TextInput style={styles.input} placeholder="Enter a task"
        value={task} onChangeText={setTask} />
      <Button title="Add Task" onPress={addTask} />
      <FlatList data={tasks}
        renderItem={({item}) => (
          <TouchableOpacity onPress={() => removeTask(item.id)}>
            <Text style={styles.taskItem}>• {item.text} (Tap to delete)</Text>
          </TouchableOpacity>
        )}
        keyExtractor={item => item.id} />
    </View>
  );
}

▾
⚙ Plugin required: cordova plugin add cordova-plugin-file

On device ready, resolves data.txt in external storage. Write button saves input text; Read button fetches it back using the FileReader API.

<!DOCTYPE html>
<html><head>
  <title>SD Card Storage</title>
  <script src="cordova.js"></script>
  <script>
    let fileEntry;
    document.addEventListener("deviceready", () => {
      window.resolveLocalFileSystemURL(cordova.file.externalDataDirectory, dir =>
        dir.getFile("data.txt", {create: true}, file => { fileEntry = file; })
      );
    }, false);

    function writeData() {
      fileEntry.createWriter(w => {
        w.write(document.getElementById('inputData').value);
        alert("Data stored in SD Card!");
      });
    }
    function readData() {
      fileEntry.file(file => {
        const r = new FileReader();
        r.onloadend = function() {
          document.getElementById('output').innerHTML = "Data: " + this.result;
        };
        r.readAsText(file);
      });
    }
  </script>
</head><body style="padding:20px;">
  <h2>SD Card File Manager</h2>
  <input type="text" id="inputData" placeholder="Enter data to save"><br><br>
  <button onclick="writeData()">Write to SD</button>
  <button onclick="readData()">Read from SD</button>
  <p id="output"></p>
</body></html>

Online Examination with DB Mock

Android Java · RadioGroup · score evaluation
▾

Evaluates an MCQ (via RadioGroup) and a fill-in-the-blank (via EditText). Tallies the score and shows a simulated "saved to DB" message.

// MainActivity.java
package com.example.onlineexam;

import android.os.Bundle;
import android.widget.*;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

        setContentView(R.layout.activity_main);

        fillBlankAnswer = findViewById(R.id.fillBlankAnswer);
        submitBtn       = findViewById(R.id.submitBtn);

        // MCQ check
        if (mcqGroup.getCheckedRadioButtonId() == R.id.correctOption)

        if (fillBlankAnswer.getText().toString().trim().equalsIgnoreCase("java"))
            score += 10;

        resultView.setText("Exam Submitted!\nYour Score: " + score + "/20");
        Toast.makeText(this, "Answers saved successfully", Toast.LENGTH_SHORT).show();
    }
}

Library Management (SQLite)

Android Java · SQLiteDatabase · INSERT · SELECT
▾

Opens/creates LibraryDB, creates an Issue table, then provides Save and View buttons to insert and read book-issue records.

// MainActivity.java
package com.example.library;

import android.content.ContentValues;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.os.Bundle;
import android.widget.*;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    SQLiteDatabase db;
    EditText       bookName, studentName;
    TextView       displayData;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        bookName    = findViewById(R.id.bookName);
        studentName = findViewById(R.id.studentName);
        displayData = findViewById(R.id.displayData);

        db = openOrCreateDatabase("LibraryDB", MODE_PRIVATE, null);
        db.execSQL("CREATE TABLE IF NOT EXISTS Issue(Book VARCHAR, Student VARCHAR);");

        findViewById(R.id.saveBtn).setOnClickListener(v -> {
            ContentValues cv = new ContentValues();
            cv.put("Book",    bookName.getText().toString());
            cv.put("Student", studentName.getText().toString());
            db.insert("Issue", null, cv);
            Toast.makeText(this, "Saved to DB!", Toast.LENGTH_SHORT).show();
        });

        findViewById(R.id.viewBtn).setOnClickListener(v -> {
            Cursor cursor = db.rawQuery("SELECT * FROM Issue", null);
            StringBuilder sb = new StringBuilder();
            while (cursor.moveToNext())
                sb.append(cursor.getString(0)).append(" issued to ")
                  .append(cursor.getString(1)).append("\n");
            displayData.setText(sb.toString());
            cursor.close();
        });
    }
}

SMS Received Alert

Android Java · BroadcastReceiver · SmsMessage
▾
⚙ Manifest: add RECEIVE_SMS permission and register SmsReceiver as a <receiver>

A BroadcastReceiver listens for SMS_RECEIVED. On arrival it extracts the sender and body from the PDU bundle and shows a Toast alert.

// SmsReceiver.java
package com.example.smsalert;

import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.os.Bundle;
import android.telephony.SmsMessage;
import android.widget.Toast;

public class SmsReceiver extends BroadcastReceiver {
    @Override
    public void onReceive(Context context, Intent intent) {
        if ("android.provider.Telephony.SMS_RECEIVED".equals(intent.getAction())) {
            Bundle bundle = intent.getExtras();
            if (bundle != null) {
                Object[] pdus = (Object[]) bundle.get("pdus");
                if (pdus != null) {
                    for (Object pdu : pdus) {
                        SmsMessage msg = SmsMessage.createFromPdu((byte[]) pdu);
                        String sender = msg.getDisplayOriginatingAddress();
                        String body   = msg.getMessageBody();
                        Toast.makeText(context,
                            "ALERT from " + sender + ": " + body,
                            Toast.LENGTH_LONG).show();
                    }
                }
            }
        }
    }
}

Payroll Processing with Leave Application

Android Java · SQLite · net-pay deduction formula
▾

Calculates Net Pay = (Basic + DA + HRA + Allowances) − (Leave days × daily rate). Result is stored in CompanyDB and displayed in a TextView.

// MainActivity.java
package com.example.payroll;

import android.content.ContentValues;
import android.database.sqlite.SQLiteDatabase;
import android.os.Bundle;
import android.widget.*;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {
    SQLiteDatabase db;
    EditText       empName, basicPay, da, hra, allowances, leaveDays;
    TextView       resultView;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        empName    = findViewById(R.id.empName);
        basicPay   = findViewById(R.id.basicPay);
        da         = findViewById(R.id.da);
        hra        = findViewById(R.id.hra);
        allowances = findViewById(R.id.allowances);
        leaveDays  = findViewById(R.id.leaveDays);
        resultView = findViewById(R.id.resultView);

        db = openOrCreateDatabase("CompanyDB", MODE_PRIVATE, null);
        db.execSQL("CREATE TABLE IF NOT EXISTS Payroll(Name VARCHAR, NetPay REAL, Leaves INTEGER);");

        findViewById(R.id.calcBtn).setOnClickListener(v -> processPayroll());
        findViewById(R.id.leaveBtn).setOnClickListener(v ->
            Toast.makeText(this,
              "Leave application for " + leaveDays.getText() +
              " days submitted for "   + empName.getText(),
              Toast.LENGTH_LONG).show()
        );
    }

    private void processPayroll() {
        try {
            double basic  = Double.parseDouble(basicPay.getText().toString());
            double daVal  = Double.parseDouble(da.getText().toString());
            double hraVal = Double.parseDouble(hra.getText().toString());
            double allow  = Double.parseDouble(allowances.getText().toString());
            int    leaves = Integer.parseInt(leaveDays.getText().toString());

            double netPay = (basic + daVal + hraVal + allow) - (leaves * basic / 30);

            ContentValues cv = new ContentValues();
            cv.put("Name",   empName.getText().toString());
            cv.put("NetPay", netPay);
            cv.put("Leaves", leaves);
            db.insert("Payroll", null, cv);

            resultView.setText(String.format("Net Pay: ₹%.2f\nSaved to DB!", netPay));
        } catch (Exception e) {
            Toast.makeText(this, "Enter valid numeric values", Toast.LENGTH_SHORT).show();
        }
    }
}

Communication Model for Media Sharing

Android Java · Intent.ACTION_SEND · image / video / audio
▾

Uses Android's implicit intent system (Intent.ACTION_SEND) to open the OS share sheet for images, videos, and audio files. Delegates actual delivery to the chosen app (Messages, Gmail, etc.).

// MainActivity.java
package com.example.mediashare;

import android.content.Intent;
import android.os.Bundle;
import android.widget.Button;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        ((Button) findViewById(R.id.shareImgBtn))  .setOnClickListener(v -> shareMedia("image/*"));
        ((Button) findViewById(R.id.shareVidBtn))  .setOnClickListener(v -> shareMedia("video/*"));
        ((Button) findViewById(R.id.shareAudioBtn)).setOnClickListener(v -> shareMedia("audio/*"));
    }

    private void shareMedia(String mimeType) {
        Intent i = new Intent(Intent.ACTION_SEND);
        i.setType(mimeType);
        i.putExtra(Intent.EXTRA_TEXT, "Sharing a " + mimeType + " file!");
        // Attach actual file URI: i.putExtra(Intent.EXTRA_STREAM, fileUri);
        startActivity(Intent.createChooser(i, "Share Media..."));
    }
}
